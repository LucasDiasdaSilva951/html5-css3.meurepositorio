Código


num = int(input('Digite um número: '))
soma = cont = 0
while True:
    num = int(input('Digite um número: '))
    if num == 0:
        break
    if num != 0:
        continue
    if num % 2 == 0:
        soma += num
        cont += 1
print(f'A soma dos {cont} números pares é {soma}!)
